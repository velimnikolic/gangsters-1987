System.Action<RoadDemo.PedestrianAgent, UnityEngine.Vector3, UnityEngine.Vector3, string> observeWrite = null;
observeWrite = (walker, from, to, source) =>
{
    try
    {
        if (done || walker.Tf == null || !walker.Tf.gameObject.activeInHierarchy) return;
        var frame = UnityEngine.Time.frameCount;
        if (!walkWrites.TryGetValue(walker, out var writes))
            walkWrites[walker] = writes = new System.Collections.Generic.List<(int, int, UnityEngine.Vector3, UnityEngine.Vector3, string)>();
        if (writes.Count > 0 && writes[0].frame != frame) writes.Clear();
        if (writes.Count < 12) writes.Add((frame, RoadDemo.WalkObstacles.Version, from, to, source));
        // InitAt calls Move(0) to place a fresh body; this is not a travelled segment.
        if (!walkPrevious.ContainsKey(walker) || source == "InitAt" ||
            walkWriteFlagged.Contains(walker) || (to - from).sqrMagnitude <= .01f) return;
        var man = walker as RoadDemo.CrewWalker;
        if (man != null && (man.Dead || man.Riding ||
            (bool)typeof(RoadDemo.CrewWalker).GetProperty("Crossing", flags).GetValue(man))) return;
        if (walker is RoadDemo.CivilianAgent civilian && (civilian.Dead || civilian.State != RoadDemo.CivilianAgent.Mode.Walking)) return;
        var radius = RoadDemo.WalkRoute.ClearanceRadius;
        if (RoadDemo.WalkObstacles.Standing(from, radius) ||
            (!RoadDemo.WalkObstacles.Standing(to, radius) && !RoadDemo.WalkObstacles.BlocksStanding(from, to, radius))) return;
        walkWriteFlagged.Add(walker);
        System.IO.File.AppendAllText(folder + "/walker-write-crossings.jsonl", Newtonsoft.Json.JsonConvert.SerializeObject(new {
            t = UnityEngine.Time.time, frame, source, geometry = RoadDemo.WalkObstacles.Version,
            kind = walker.GetType().Name, id = man?.CharacterId ?? 0, from = from.ToString(), to = to.ToString(),
            state = man?.State.ToString(), door = man == null ? null : RoadDemo.DoorBeat.PhaseOf(man).ToString()
        }) + "\n");
    }
    catch (System.Exception error) { Finish("FAIL", "Individual walk-write observer failed: " + error); }
};
RoadDemo.PedestrianAgent.WalkStepObserved += observeWrite;
void ObserveWalks(float now)
{
    if (now == lastWalkTime) return;
    lastWalkTime = now;
    walkSamples++;
    var walkers = crews.Units.SelectMany(u => u.All()).Cast<RoadDemo.PedestrianAgent>()
        .Concat(RoadDemo.CivilianAgent.All).Distinct();
    foreach (var walker in walkers)
    {
        bool visible = walker.Tf != null && walker.Tf.gameObject.activeInHierarchy;
        bool walking = walker is RoadDemo.CrewWalker crewWalker
            ? !crewWalker.Dead && !crewWalker.Riding && !RoadDemo.DoorBeat.Active(crewWalker)
            : walker is RoadDemo.CivilianAgent civilian && !civilian.Dead && civilian.State == RoadDemo.CivilianAgent.Mode.Walking;
        if (walker.Tf != null)
        {
            var at = walker.Tf.position;
            var man = walker as RoadDemo.CrewWalker;
            var door = man == null ? null : RoadDemo.DoorBeat.PhaseOf(man).ToString();
            bool hadPrevious = bodyPrevious.TryGetValue(walker, out var previous);
            if (visible && walking && RoadDemo.WalkObstacles.Standing(at, .1f) && embeddedFlagged.Add(walker))
                System.IO.File.AppendAllText(folder + "/embedded-walkers.jsonl", Newtonsoft.Json.JsonConvert.SerializeObject(new {
                    t = now, frame = UnityEngine.Time.frameCount, id = man?.CharacterId ?? 0,
                    kind = walker.GetType().Name, at = at.ToString(), door, geometry = RoadDemo.WalkObstacles.Version,
                    hadPrevious, previous = new { at = previous.position.ToString(), previous.visible, previous.door, previous.geometry },
                    writes = walkWrites.TryGetValue(walker, out var journal) ? journal.Select(w => new {
                        w.frame, w.geometry, w.source, from = w.from.ToString(), to = w.to.ToString() }).ToArray() : null
                }) + "\n");
            bodyPrevious[walker] = (at, visible, door, RoadDemo.WalkObstacles.Version);
        }
        if (!visible || !walking) { walkPrevious.Remove(walker); continue; }
        var position = walker.Tf.position;
        if (walkPrevious.TryGetValue(walker, out var from) && (position - from).sqrMagnitude > 0.01f &&
            !RoadDemo.WalkObstacles.Standing(from, RoadDemo.WalkRoute.ClearanceRadius) &&
            !RoadDemo.WalkObstacles.Standing(position, RoadDemo.WalkRoute.ClearanceRadius))
        {
            checkedWalkSegments++;
            if (!walkFlagged.Contains(walker) && RoadDemo.WalkObstacles.BlocksStanding(from, position, RoadDemo.WalkRoute.ClearanceRadius))
            {
                walkFlagged.Add(walker);
                var nearby = new System.Collections.Generic.List<RoadDemo.SidewalkPlan.Box>();
                RoadDemo.WalkObstacles.PropsNear((from + position) * .5f, (position - from).magnitude + 1f, nearby);
                var touched = nearby.Where(box => {
                    var shape = new RoadDemo.SidewalkPlan(); shape.Take(box);
                    return shape.Obstructs(new UnityEngine.Vector2(from.x, from.z), new UnityEngine.Vector2(position.x, position.z), RoadDemo.WalkRoute.ClearanceRadius);
                }).Select(box => new { box.SourceName, centre = box.C.ToString(), half = box.H.ToString(), ax = box.Ax.ToString(), az = box.Az.ToString() }).ToArray();
                var crewMan = walker as RoadDemo.CrewWalker;
                System.IO.File.AppendAllText(folder + "/walker-crossings.jsonl", Newtonsoft.Json.JsonConvert.SerializeObject(new {
                    t = now, kind = walker.GetType().Name, id = walker is RoadDemo.CrewWalker man ? man.CharacterId : 0,
                    from = from.ToString(), to = position.ToString(), distance = (position - from).magnitude,
                    frame = UnityEngine.Time.frameCount, dt = UnityEngine.Time.deltaTime, touched,
                    geometryVersion = RoadDemo.WalkObstacles.Version, previousGeometryVersion = lastWalkVersion,
                    writes = walkWrites.TryGetValue(walker, out var recent) ? recent.Select(w => new {
                        w.frame, w.geometry, w.source, from = w.from.ToString(), to = w.to.ToString() }).ToArray() : null,
                    state = crewMan?.State.ToString(), crossing = crewMan != null && (bool)typeof(RoadDemo.CrewWalker).GetProperty("Crossing", flags).GetValue(crewMan),
                    goal = crewMan?.OrderDestination.ToString(),
                    legs = crewMan == null ? null : ((System.Collections.Generic.List<UnityEngine.Vector3>)typeof(RoadDemo.CrewWalker).GetField("_legs", flags).GetValue(crewMan)).Select(at => at.ToString()).ToArray()
                }) + "\n");
            }
        }
        walkPrevious[walker] = position;
    }
    lastWalkVersion = RoadDemo.WalkObstacles.Version;
}
void Finish(string status, string reason)
{
    if (done) return;
    done = true;

                    if (!RoadDemo.DoorBeat.Active(man) && RoadDemo.WalkObstacles.Standing(man.Tf.position, .225f))
                    { Finish("FAIL", "New recruit appeared inside a blocked structure."); return; }
                }
                if (!RoadDemo.CrewQuarters.Inside(unit) || RoadDemo.DoorBeat.PhaseOf(man) != RoadDemo.DoorBeat.VisitPhase.Inside) return;
                if (!crews.OrderUnit(unit, unit.Position + UnityEngine.Vector3.right * 18f, out moveDestination))
                { Finish("FAIL", "Newly enlarged indoor crew could not move outside."); return; }
                roleStep = 3;
            }
            else if (roleStep == 3 && (unit.Position - moveDestination).sqrMagnitude < 4f &&
                unit.All().All(m => m.Tf != null && m.Tf.gameObject.activeInHierarchy && !RoadDemo.DoorBeat.Active(m)))
            {
                var faults = new System.Collections.Generic.List<string>(); p.ValidateOrganization(faults);
                Finish(faults.Count == 0 ? "PASS" : "FAIL", faults.Count == 0
                    ? "Recruited beside an indoor lieutenant, entered with the crew, then all bodies walked outside."
                    : string.Join("; ", faults));
            }
            return;
        }
        if (kind == "streaming")
        {
            if (roleStep == 0)
            {
                var home = LivingCity.Business.CityBusinesses.All.First(b => LivingCity.Gameplay.DoorHolder.Read(b.Id) == LivingCity.Outfit.DoorTenure.Ours);
                if (!RoadDemo.CrewQuarters.Station(crews, unit, home.Id))
                { Finish("FAIL", "Could not station crew before inspecting the city's streamed blocks."); return; }
                roleStep = 1;
            }
            var recycler = UnityEngine.Object.FindAnyObjectByType<RoadDemo.CityBlockRecycler>();
            var model = (RoadDemo.ResidentialBlockModel)recycler.GetType().GetField("_model", flags).GetValue(recycler);
            if (recycler.NavigationBlocks != model.Count)
            { Finish("FAIL", "Walkers started before every block had navigation geometry."); return; }
            if (streamIndex >= model.Count)
            {
                Log("navigation bake " + recycler.NavigationBakeMs + "ms; rendered visits " + streamVisited.Count + "; evictions " + recycler.EvictedViews);
                Finish(recycler.EvictedViews > 0 ? "PASS" : "NOT_EXERCISED",
                    "Visited every residential block; rendered collision matched the persistent plan through visual evictions.");
                return;
            }
            var recipe = model.Blocks[(streamIndex * (model.Count - 1)) % model.Count];
            if (streamAt < 0)
            {
                var rig = UnityEngine.Object.FindAnyObjectByType<RoadDemo.DemoCamera>();
                var target = recycler.transform.TransformPoint(new UnityEngine.Vector3(recipe.LocalBounds.center.x, 0, recipe.LocalBounds.center.y));
                rig.PanBy(new UnityEngine.Vector2(target.x - rig.pivot.x, target.z - rig.pivot.z));
                streamAt = now; Log("pan to " + recipe.Id);
            }
            var resident = (System.Collections.IDictionary)recycler.GetType().GetField("_resident", flags).GetValue(recycler);
            if (resident.Contains(recipe.Id))
            {
                var view = resident[recipe.Id];
                var content = (UnityEngine.Transform)view.GetType().GetField("Content").GetValue(view);
                var uncovered = RoadDemo.WalkObstacles.ComposedPropPlan(content, recycler.transform.position.y);
                if (uncovered.Count > 0)
                {
                    System.IO.File.WriteAllText(folder + "/unplanned-props.json", Newtonsoft.Json.JsonConvert.SerializeObject(
                        uncovered.Boxes.Select(b => new { b.SourceName, centre = b.C.ToString(), half = b.H.ToString() })));
                    Finish("FAIL", "The streamed block introduced geometry absent from the startup plan: " + recipe.Id); return;
                }
                streamVisited.Add(recipe.Id); streamIndex++; streamAt = -1f;
            }
            else if (now - streamAt > 600f)
            { Finish("FAIL", "Camera reached a block whose visual payload never completed: " + recipe.Id); }
            return;
        }
        if (kind == "roles")
        {
            var bag = crews.Units.FirstOrDefault(u => u.Faction == 0 && u.CrewId == crewId && u.IsDetachment);
            if (roleStep == 0 && bag != null && RoadDemo.CrewQuarters.Inside(bag))
            {
                var pulled = p.PullEscort(2); var swap = p.NameCollector(crewId, 2); var escort = p.PostEscort(crewId, 5);
                Log($"swap indoor bag: pull {pulled.Ok}; collector {swap.Ok}; escort {escort.Ok}");
                if (!pulled.Ok || !swap.Ok || !escort.Ok) { Finish("FAIL", "Indoor duty change refused."); return; }
                roleStep = 1; next = now + 8f;
            }
            else if (roleStep == 1 && now > next && bag != null && RoadDemo.CrewQuarters.Inside(bag))
            {
                var promoted = p.Promote(5, out promotedCrew);
                Log("promote indoor escort " + promoted.Ok + " " + promoted.Reason);
                if (!promoted.Ok) { Finish("FAIL", "Indoor escort promotion refused: " + promoted.Reason); return; }
                roleStep = 2; next = now + 2f;
            }
            else if (roleStep == 2 && now > next)
            {
                var promoted = crews.Units.FirstOrDefault(u => u.Faction == 0 && u.CrewId == promotedCrew && !u.IsDetachment);
                if (promoted == null) return;
                var moved = crews.OrderUnit(promoted, promoted.Position + UnityEngine.Vector3.left * 12f, out moveDestination);
                Log("move promoted escort " + moved);
                if (!moved) { Finish("FAIL", "Promoted escort refused movement."); return; }
                roleStep = 3;
            }
            else if (roleStep == 3)
            {
                var promoted = crews.Units.FirstOrDefault(u => u.Faction == 0 && u.CrewId == promotedCrew && !u.IsDetachment);
                if (promoted?.Boss?.Tf != null && promoted.Boss.Tf.gameObject.activeInHierarchy &&
                    !RoadDemo.DoorBeat.Held(promoted.Boss) && (promoted.Position - moveDestination).sqrMagnitude < 4f)
                {
                    var faults = new System.Collections.Generic.List<string>(); p.ValidateOrganization(faults);
                    Finish(faults.Count == 0 ? "PASS" : "FAIL", faults.Count == 0
                        ? "Indoor collector/escort swap and escort promotion preserved a visible, movable body and valid organization."
                        : string.Join("; ", faults));
                }
            }
            return;
        }
        if (unit.Wiped)
        {
            if (kind == "fight" && challengeAt >= 0)
            {
                if (defeatAt < 0) { defeatAt = now; Log("crew defeated; watching corpse cleanup"); }
                if (crews.OrderUnit(unit, unit.Position + UnityEngine.Vector3.right * 10f, out _))
                { Finish("FAIL", "A wiped crew still accepted movement."); return; }
                if (now - defeatAt > 120f && !dispatch.Swarming)
                    Finish(everSwarmed ? "PASS" : "NOT_EXERCISED", everSwarmed
                        ? "Defeated crew rejects orders after corpse cleanup and swarm has cleared."
                        : "Crew was defeated but this run never exercised a swarm.");
            }
            else Finish("FAIL", "The trial crew was wiped out before the requested outcome.");
            return;
        }
        var suspect = Read("_arrestCrew") as RoadDemo.DemoCrews.Unit;
        // A collection player waits for the encounter already taking place. Sending
        // the lieutenant indoors during the question is itself an ordered escape.
        if (collecting && suspect != null && suspect.Faction == 0 && suspect.CrewId == crewId)
            return;
        if ((legal || kind == "arrest" || kind == "flee" || kind == "fight" || kind == "fight_win") &&
            suspect != null && suspect.Faction == 0 && suspect.CrewId == crewId)
        {
            var phase = Read("_collar").ToString();
            var state = phase + " " + Read("_answer");
            if (lastPolice != state) { Log("arrest " + state); lastPolice = state; }
            if ((kind == "fight" || kind == "fight_win") && answeredFightAt != (float)Read("_collarAt"))
            {
                if (kind == "fight_win" && victoryAt >= 0)
                {
                    var home = LivingCity.Business.CityBusinesses.All.Where(b => LivingCity.Gameplay.DoorHolder.Read(b.Id) == LivingCity.Outfit.DoorTenure.Ours)
                        .OrderBy(b => rt.TryGetBusinessApproach(b.Id, out var at) ? (at - unit.Position).sqrMagnitude : float.MaxValue).First();
                    if (!RoadDemo.CrewQuarters.Station(crews, unit, home.Id)) { Finish("FAIL", "Could not reaffirm shelter during a new police challenge."); return; }
                    answeredFightAt = (float)Read("_collarAt");
                    Log("player kept the shelter order during the new police challenge");
                    return;
                }
                var officer = Read("_arrestOfficer") as RoadDemo.PoliceBeat;
                var lawNow = Read("_arrestSquad") as RoadDemo.DemoCrews.Unit ?? officer?.Unit;
                crews.Select(unit);
                if (lawNow == null || !(escapeAt >= 0 && kind == "fight_win"
                    ? crews.OrderFlee(unit, lawNow.Position) : crews.OrderAttack(lawNow)))
                { Finish("FAIL", "Attack during the police challenge was refused."); return; }
                answeredFightAt = (float)Read("_collarAt");
                challengeAt = challengeAt < 0 ? now : challengeAt;
                challengedBy = lawNow;
                Log(escapeAt >= 0 && kind == "fight_win" ? "withdrawal reaffirmed during the police challenge" : "attack reaffirmed during the police challenge");
            }
            if (challengeAt < 0)
            {
                challengeAt = now;
                if ((legal || kind == "arrest") && Read("_answer").ToString() != "Quiet")
                { Finish("NOT_EXERCISED", "The lieutenant naturally chose " + Read("_answer")); return; }
                var beat = Read("_arrestOfficer") as RoadDemo.PoliceBeat;
                var law = Read("_arrestSquad") as RoadDemo.DemoCrews.Unit ?? beat?.Unit;
                challengedBy = law;
                Log("challenge positions: crew " + unit.Position + "; police " + (law != null ? law.Position.ToString() : "missing") +
                    "; distance " + (law != null ? UnityEngine.Vector3.Distance(unit.Position, law.Position) : -1f));
                if (kind == "flee")
                {
                    escapeFrom = unit.Position;
                    var awayFrom = law != null ? law.Position : unit.Position + UnityEngine.Vector3.right;
                    var accepted = crews.OrderFlee(unit, awayFrom);
                    Log("flee order " + accepted);
                    if (!accepted) { Finish("FAIL", "Flee order was refused during the challenge."); return; }
                    escapeAt = now;
                }
                if (kind == "fight" || kind == "fight_win")
                {
                    crews.Select(unit);
                    if (law == null || !crews.OrderAttack(law))
                    { Finish("FAIL", "Attack on arresting police was refused."); return; }
                    Log("attack arresting police");
                }
            }
            return;
        }
        if (challengeAt >= 0)
        {
            if ((kind == "fight" || kind == "fight_win") && victoryAt < 0 && escapeAt < 0 &&
                unit.Fleeing && challengedBy != null && !challengedBy.Wiped)
            {
                crews.Select(unit);
                if (!crews.OrderAttack(challengedBy)) { Finish("FAIL", "Could not countermand automatic flight with an attack."); return; }
                Log("player countermanded automatic flight and resumed the attack");
            }
            if (kind == "flee" && (unit.Position - escapeFrom).sqrMagnitude > 20f * 20f)
                escapedDistance = true;
            if (kind == "flee" && escapeAt >= 0 && now - escapeAt > 20f &&
                escapedDistance && p.Roster.Find(leader).WantedLevel > 0)
            {
                if (!hidAfterFight && RoadDemo.CrewQuarters.Billeted(unit))
                {
                    hidAfterFight = true;
                    Log("fugitives broke sight and are following the normal shelter decision");
                }
                if (hidAfterFight && RoadDemo.CrewQuarters.Inside(unit) && !dispatch.Swarming &&
                    unit.All().Where(m => !m.Dead).All(m => m.Tf != null &&
                        !m.Tf.gameObject.activeInHierarchy && RoadDemo.DoorBeat.PhaseOf(m) == RoadDemo.DoorBeat.VisitPhase.Inside))
                { Finish("PASS", "Crew fled, became wanted, physically reached shelter and cleared pursuit."); return; }
            }
            if (kind == "fight_win" && (victoryAt >= 0 || challengedBy != null && challengedBy.Wiped))
            {
                if (victoryAt < 0) { victoryAt = now; Log("arresting police killed"); }
                if (!hidAfterFight)
                {
                    var home = LivingCity.Business.CityBusinesses.All.Where(b => LivingCity.Gameplay.DoorHolder.Read(b.Id) == LivingCity.Outfit.DoorTenure.Ours)
                        .OrderBy(b => rt.TryGetBusinessApproach(b.Id, out var at) ? (at - unit.Position).sqrMagnitude : float.MaxValue).First();
                    hidAfterFight = RoadDemo.CrewQuarters.Station(crews, unit, home.Id);
                    if (!hidAfterFight) { Finish("FAIL", "Survivors could not return home after defeating the patrol."); return; }
                    Log("survivors ordered into the nearest owned building immediately after defeating the patrol");
                }
                hidAfterFight |= RoadDemo.CrewQuarters.Billeted(unit);
                if (hidAfterFight && everSwarmed && !dispatch.Swarming && RoadDemo.CrewQuarters.Inside(unit) &&
                    unit.All().Where(m => !m.Dead).All(m => m.Tf != null &&
                        !m.Tf.gameObject.activeInHierarchy && RoadDemo.DoorBeat.PhaseOf(m) == RoadDemo.DoorBeat.VisitPhase.Inside))
                { Finish("PASS", "Arresting police killed; surviving crew reached shelter and the triggered swarm cleared."); return; }
            }
            if (now - challengeAt > 900f) Finish("FAIL", "Arrest/flight/combat did not complete after the challenge.");
            return;
        }
        if (kind == "racket_rendered" || kind == "racket_overlap" || kind == "arson")
        {
            if (!renderedReady)
            {
                if (renderedTargets.Count == 0)
                {
                    renderedTargets.AddRange(LivingCity.Business.CityBusinesses.All.Where(b => rt.IsRacketable(b.Id) &&
                        LivingCity.Gameplay.DoorHolder.Read(b.Id) != LivingCity.Outfit.DoorTenure.Ours &&
                        !RoadDemo.ShopDamage.IsBusinessDamaged(b.Id))
                        .OrderBy(b => rt.TryGetBusinessApproach(b.Id, out var at) ? (at - unit.Position).sqrMagnitude : float.MaxValue)
                        .Take(kind == "arson" ? 1 : 3).Select(b => b.Id));
                    if (renderedTargets.Count == 0) { Finish("FAIL", "No business for the rendered doorway scenario."); return; }
                    rt.TryGetBusinessApproach(renderedTargets[0], out var atDoor);
                    var rig = UnityEngine.Object.FindAnyObjectByType<RoadDemo.DemoCamera>();
                    rig.PanBy(new UnityEngine.Vector2(atDoor.x - rig.pivot.x, atDoor.z - rig.pivot.z));
                    Log("camera moved to the target block; waiting for physical business objects");
                }
                foreach (var id in renderedTargets)
                    if (!LivingCity.Business.BusinessViewBindings.TryGet(id, out var marker) || marker == null ||
                        !marker.gameObject.activeInHierarchy || marker.GetComponentsInChildren<UnityEngine.Renderer>(true).Length == 0)
                        return;
                foreach (var id in renderedTargets)
                {
                    LivingCity.Business.BusinessViewBindings.TryGet(id, out var marker);
                    var front = marker.GetComponent<RoadDemo.Storefront>() ?? marker.GetComponentInParent<RoadDemo.Storefront>();
                    Log("physical doorway ready " + id + "; storefront leaves " + (front != null ? front.LeafCount : -1));
                }
                renderedReady = true;
            }
            if (kind == "arson")
            {
                var targetDoor = renderedTargets[0];
                if (roleStep == 0)
                {
                    if (!LivingCity.Gameplay.DoorJobs.TryBuild(targetDoor, LivingCity.Outfit.OrderType.Torch,
                            crewId, UnityEngine.Mathf.Max(1, unit.Standing()), out var job, out var refusal))
                    { Finish("FAIL", "Torch refused: " + refusal); return; }
                    var issued = LivingCity.Gameplay.OutfitDirector.Instance.IssueOrder(job);
                    Log("torch order " + issued.Ok + " " + issued.Reason);
                    if (!issued.Ok) { Finish("FAIL", "Torch job was refused: " + issued.Reason); return; }
                    roleStep = 1; next = now + 180f;
                }
                else if (LivingCity.Business.BusinessRuntime.Instance.TryGetShutdown(targetDoor, out var closure) &&
                    closure.Cause == LivingCity.Business.BusinessShutdownCause.Arson &&
                    LivingCity.Gameplay.OutfitDirector.Instance.Book.Jobs.Count == 0)
                    Finish(RoadDemo.ShopDamage.IsBusinessDamaged(targetDoor) ? "PASS" : "FAIL",
                        "Player torch job completed and the loaded business is closed by arson with damage recorded.");
                else if (now > next) Finish("FAIL", "Filed torch did not close the loaded business.");
                return;
            }
        }
        if (kind == "beating" || kind == "killing" || kind == "racket" || kind == "racket_rendered" || kind == "racket_overlap")
        {
            if (probe == null)
            {
                probe = dispatch.gameObject.AddComponent<RoadDemo.RacketProbe>();
                probe.patience = 150f; probe.startAfter = 0f;
                probe.beating = kind == "beating"; probe.killing = kind == "killing";
                probe.overlap = kind == "racket_overlap";
                typeof(RoadDemo.RacketProbe).GetField("_ours", flags).SetValue(probe, unit);
                Log("started production doorstep probe for crew " + crewId);
            }
            if (probe.Finished)
            {
                if (!probe.Verdict.StartsWith("no fault")) { Finish("FAIL", probe.Verdict); return; }
                if (RoadDemo.DoorBeat.Active(unit.Boss)) return;
                if (!moveSent)
                {
                    moveSent = crews.OrderUnit(unit, unit.Position + UnityEngine.Vector3.right * 10f, out moveDestination);
                    Log("walk after completed proprietor action " + moveSent + " destination " + moveDestination);
                    if (!moveSent) Finish("FAIL", "The proprietor action completed but its crew could not walk away: " + crews.OrderRefusal);
                }
                else if ((unit.Position - moveDestination).sqrMagnitude < 4f &&
                    unit.All().Where(m => !m.Dead).All(m => m.Tf != null && m.Tf.gameObject.activeInHierarchy))
                    Finish("PASS", probe.Verdict + "\nPhysical doorway exit and subsequent player movement completed.");
            }
            return;
        }
        if (collecting && banked > 0 && requestedCollectionBanked)
        {
            var money = LivingCity.Gameplay.OutfitDirector.Instance.Accounts;
            if (!physicallyPaid || !bankedAtHeadquarters || money.Sheets.Sum(s => s.IllegalIncome) - initialIllegal != banked ||
                money.Safe - initialCash != LedgerCash() - initialLedgerCash)
                Finish("FAIL", $"Banking {banked} failed physical collection or transaction reconciliation.");
            else Finish("PASS", $"Collector returned {banked} after physical door visits; receipts and safe reconcile including daily expenses.");
            return;
        }
        if (kind == "doorstep" && answered >= 3)
        {
            if (RoadDemo.DoorBeat.Active(unit.Boss)) return;
            if (!moveSent)
            {
                moveSent = crews.OrderUnit(unit, unit.Position + UnityEngine.Vector3.right * 10f, out moveDestination);
                Log("move after third doorway " + moveSent);
                if (!moveSent) Finish("FAIL", "Crew could not move after exiting third shop.");
            }
            else if ((unit.Position - moveDestination).sqrMagnitude < 4f && unit.All().All(m => m.Tf != null && m.Tf.gameObject.activeInHierarchy))
                Finish("PASS", "Three owner answers, physical exits, and a subsequent move completed.");
            return;
        }
        if (now < next || collectionSent) return;
        if (waiting)
        {
            var count = rt.Racket.TryGetRelationship(pending, LivingCity.Gameplay.PlayerCommands.House, out var rel)
                ? rel.Demands : 0;
            if (count > beforeDemands) { answered++; waiting = false; Log("door answered " + rel.State); next = now + 4f; }
            else if (now - pendingAt > 120f) { Finish("FAIL", "Accepted demand never reached an owner answer: " + pending); }
            return;
        }
        var protectedDoors = collecting ? LivingCity.Business.CityBusinesses.All.Where(b =>
            visited.Contains(b.Id.Value) && rt.Racket.TryGetRelationship(b.Id, LivingCity.Gameplay.PlayerCommands.House, out var relation) &&
            relation.IsProtected).OrderByDescending(b => rt.Dues.OwedOf(b.Id, LivingCity.Gameplay.PlayerCommands.House)).ToArray() : null;
        if (collecting && answered >= 1 && protectedDoors.Length > 0)
        {
            var row = protectedDoors[0];
            pending = row.Id;
            if (collectionBlocks.Add(row.CanonicalBlockId.Value))
            {
                var assignment = p.AssignBlockResponsibility(row.CanonicalBlockId, leader);
                Log("block assignment " + assignment.Ok + " " + assignment.Reason);
                if (!assignment.Ok) { Finish("FAIL", "Block responsibility refused: " + assignment.Reason); return; }
                var blockSource = LivingCity.UI.BlockRacketSeam.Source;
                if (blockSource == null || !blockSource.TryGetBlock(row.CanonicalBlockId, out var blockView) ||
                    !blockView.HasResponsible || blockView.ResponsibleCrewId != crewId ||
                    blockView.CollectorId != 5 || blockView.Collectors != 1 ||
                    !blockSource.IsCollector(5) || blockSource.IsCollector(2))
                { Finish("FAIL", "Block details disagree with the assigned collector or label his escort as a second collector."); return; }
                Log("block details: collector " + blockView.CollectorId + " " + blockView.CollectorName +
                    "; count " + blockView.Collectors + "; schedule " + blockView.CollectsWord);
                blockAssigned = true;
                var home = LivingCity.Business.CityBusinesses.All.First(b => LivingCity.Gameplay.DoorHolder.Read(b.Id) == LivingCity.Outfit.DoorTenure.Ours);
                Log("line returns indoors " + RoadDemo.CrewQuarters.Station(crews, unit, home.Id));
            }
            if (!RoadDemo.CrewQuarters.Billeted(unit) && !RoadDemo.DoorBeat.Active(unit.Boss))
            {
                var home = LivingCity.Business.CityBusinesses.All.First(b => LivingCity.Gameplay.DoorHolder.Read(b.Id) == LivingCity.Outfit.DoorTenure.Ours);
                Log("line returns indoors between collections " + RoadDemo.CrewQuarters.Station(crews, unit, home.Id));
            }
            if (!rt.Dues.TryGet(pending, out var due) || due.Owed <= 0 || rt.GameHour % 24 < 9 || rt.GameHour % 24 > 17)
            { clock.SetSpeed(4); next = now + 10f; return; }
            if (kind == "collection_auto") { next = now + 10f; return; }
            if ((int)(rt.GameHour / 24d) < retryCollectionDay) { next = now + 10f; return; }
            clock.SetSpeed(4);
            safeBefore = LivingCity.Gameplay.OutfitDirector.Instance.Accounts.Safe;
            var sent = rt.Commands.Submit(LivingCity.Gameplay.PlayerCommands.Stamp(new LivingCity.Territory.CollectDuesCommand(
                LivingCity.Territory.TerritoryCommandNodeId.Crew(crewId), row.CanonicalBlockId)));
            Log("collection " + sent.Status + " " + sent.Reason);
            if (!sent.WasAccepted) { Finish("FAIL", "Dues exist but collection refused: " + sent.Reason); return; }
            collectionSent = true;
            return;
        }
        if (!collecting && kind != "doorstep" && now - calledAt < 600f) return;
        if (RoadDemo.DoorBeat.Active(unit.Boss) && !RoadDemo.CrewQuarters.Inside(unit)) return;
        if (++attempts > 20) { Finish("FAIL", "Twenty demands did not produce the requested police scenario."); return; }
        var candidates = LivingCity.Business.CityBusinesses.All.Where(b => rt.IsRacketable(b.Id) && !visited.Contains(b.Id.Value) &&
            LivingCity.Gameplay.DoorHolder.Read(b.Id) != LivingCity.Outfit.DoorTenure.Ours &&
            !RoadDemo.ShopDamage.IsBusinessDamaged(b.Id)).OrderBy(b =>
                collecting && b.Name == "Grimaldi Grocery" ? -10000f : rt.TryGetBusinessApproach(b.Id, out var at) ?
                    (at - unit.Position).magnitude + (100f - rt.OwnerProfileOf(b.Id).Connections) * (collecting ? 0f : 0.5f)
                    : float.MaxValue).ToArray();
        if (candidates.Length == 0) { Finish("FAIL", "No accessible target businesses remain."); return; }
        var targetChoice = candidates[policeScenario && attempts == 1
            ? (attempt - 1) % System.Math.Min(3, candidates.Length) : 0];
        pending = targetChoice.Id; visited.Add(pending.Value);
        beforeDemands = rt.Racket.TryGetRelationship(pending, LivingCity.Gameplay.PlayerCommands.House, out var previous) ? previous.Demands : 0;
        var demand = rt.Commands.Submit(LivingCity.Gameplay.PlayerCommands.Stamp(new LivingCity.Territory.ApproachBusinessCommand(
            LivingCity.Territory.TerritoryCommandNodeId.Crew(crewId), pending, LivingCity.Territory.TerritoryRacketIntent.Demand)));
        Log("demand " + targetChoice.Name + " " + demand.Status + " " + demand.Reason);
        if (!demand.WasAccepted) { Finish("FAIL", "Demand order rejected: " + targetChoice.Name + " " + demand.Reason); return; }
        waiting = demand.WasAccepted; pendingAt = now; next = now + 2f;
    }
    catch (System.Exception ex) { Finish("FAIL", ex.ToString()); }
};
// Sample after every runtime frame; editor callbacks may skip multiple frames at 16x.
walkLoop = () => { try { if (!done && UnityEditor.EditorApplication.isPlaying) { ObserveWalks(UnityEngine.Time.time); tick(); } } catch (System.Exception error) { Finish("FAIL", "Runtime player driver failed: " + error); } };
var observedLoop = UnityEngine.LowLevel.PlayerLoop.GetCurrentPlayerLoop();
observedLoop.subSystemList = observedLoop.subSystemList.Concat(new[] {
    new UnityEngine.LowLevel.PlayerLoopSystem { type = typeof(RoadDemo.DriveTrace), updateDelegate = walkLoop }
}).ToArray();
UnityEngine.LowLevel.PlayerLoop.SetPlayerLoop(observedLoop);
clock.SetSpeed(4);
clock.Paused = false;
return new { kind, attempt, crewId, leader, folder };
